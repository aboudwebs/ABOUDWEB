-- ---------------------------------------------------------------------
-- 1) المسؤولون المصرّح لهم بإدارة التطبيقات
--    (تسجيل الدخول الحالي كما هو؛ هذا الجدول يحدد من يملك حق الكتابة فقط)
-- ---------------------------------------------------------------------
create table if not exists public.admin_users (
  user_id    uuid primary key references auth.users(id) on delete cascade,
  created_at timestamptz not null default now()
);
alter table public.admin_users enable row level security;

drop policy if exists "admin_users_read_own" on public.admin_users;
create policy "admin_users_read_own" on public.admin_users
  for select to authenticated using (user_id = auth.uid());
grant select on public.admin_users to authenticated;

create or replace function public.is_aboud_admin()
returns boolean
language sql
stable
security definer
set search_path = public
as $$
  select exists (select 1 from public.admin_users where user_id = auth.uid());
$$;
revoke all on function public.is_aboud_admin() from public;
grant execute on function public.is_aboud_admin() to anon, authenticated;

-- ⚠️ خطوة إلزامية: ضع بريد حساب الأدمن الذي تدخل به إلى لوحة التحكم مكان النص التالي.
--    إن تركته كما هو فلن يُضاف أحد، ولن يستطيع أي حساب تعديل التطبيقات (الوضع الآمن).
insert into public.admin_users (user_id)
select id from auth.users
where lower(email) in (lower('aboudwebs@gmail.com'))
on conflict do nothing;

-- ---------------------------------------------------------------------
-- 2) جدول التطبيقات
-- ---------------------------------------------------------------------
create table if not exists public.apps (
  id              uuid primary key default gen_random_uuid(),
  slug            text not null unique,
  name_ar         text not null,
  name_en         text not null default '',
  short_desc_ar   text not null default '',
  short_desc_en   text not null default '',
  long_desc_ar    text not null default '',
  long_desc_en    text not null default '',
  features_ar     jsonb not null default '[]'::jsonb,
  features_en     jsonb not null default '[]'::jsonb,
  icon_url        text,
  screenshots     jsonb not null default '[]'::jsonb,
  status          text not null default 'coming_soon',
  is_published    boolean not null default false,
  sort_order      integer not null default 0,
  version         text,
  apk_url         text,
  google_play_url text,
  app_store_url   text,
  external_url    text,
  created_at      timestamptz not null default now(),
  updated_at      timestamptz not null default now(),

  constraint apps_slug_format  check (slug ~ '^[a-z0-9]+(-[a-z0-9]+)*$' and char_length(slug) <= 60),
  constraint apps_status_valid check (status in ('available', 'in_development', 'coming_soon')),
  constraint apps_json_arrays  check (
    jsonb_typeof(features_ar) = 'array' and jsonb_typeof(features_en) = 'array'
    and jsonb_typeof(screenshots) = 'array' and jsonb_array_length(screenshots) <= 8
  ),
  constraint apps_lengths check (
    char_length(name_ar) <= 80 and char_length(name_en) <= 80
    and char_length(short_desc_ar) <= 280 and char_length(short_desc_en) <= 280
    and char_length(long_desc_ar) <= 5000 and char_length(long_desc_en) <= 5000
    and (version is null or char_length(version) <= 40)
  ),
  -- الروابط https فقط (يمنع javascript: وغيره)
  constraint apps_urls_https check (
    (icon_url        is null or icon_url        ~* '^https://') and
    (apk_url         is null or apk_url         ~* '^https://') and
    (google_play_url is null or google_play_url ~* '^https://') and
    (app_store_url   is null or app_store_url   ~* '^https://') and
    (external_url    is null or external_url    ~* '^https://')
  )
);

create index if not exists apps_published_order_idx on public.apps (is_published, sort_order);

create or replace function public.aboud_touch_updated_at()
returns trigger
language plpgsql
set search_path = public
as $$
begin
  new.updated_at = now();
  return new;
end;
$$;

drop trigger if exists apps_touch_updated_at on public.apps;
create trigger apps_touch_updated_at
  before update on public.apps
  for each row execute function public.aboud_touch_updated_at();

alter table public.apps enable row level security;

-- الزائر يرى المنشور فقط. المسؤول يرى الكل.
drop policy if exists "apps_public_read" on public.apps;
create policy "apps_public_read" on public.apps
  for select to anon, authenticated
  using (is_published = true or public.is_aboud_admin());

drop policy if exists "apps_admin_insert" on public.apps;
create policy "apps_admin_insert" on public.apps
  for insert to authenticated with check (public.is_aboud_admin());

drop policy if exists "apps_admin_update" on public.apps;
create policy "apps_admin_update" on public.apps
  for update to authenticated
  using (public.is_aboud_admin()) with check (public.is_aboud_admin());

drop policy if exists "apps_admin_delete" on public.apps;
create policy "apps_admin_delete" on public.apps
  for delete to authenticated using (public.is_aboud_admin());

grant select on public.apps to anon, authenticated;
grant insert, update, delete on public.apps to authenticated;

create table if not exists public.apps_page_settings (
  id         integer primary key default 1 check (id = 1),
  data       jsonb not null default '{}'::jsonb check (jsonb_typeof(data) = 'object'),
  updated_at timestamptz not null default now()
);

insert into public.apps_page_settings (id, data) values (1, '{}'::jsonb)
on conflict (id) do nothing;

drop trigger if exists apps_page_touch_updated_at on public.apps_page_settings;
create trigger apps_page_touch_updated_at
  before update on public.apps_page_settings
  for each row execute function public.aboud_touch_updated_at();

alter table public.apps_page_settings enable row level security;

drop policy if exists "apps_page_public_read" on public.apps_page_settings;
create policy "apps_page_public_read" on public.apps_page_settings
  for select to anon, authenticated using (true);

drop policy if exists "apps_page_admin_insert" on public.apps_page_settings;
create policy "apps_page_admin_insert" on public.apps_page_settings
  for insert to authenticated with check (public.is_aboud_admin());

drop policy if exists "apps_page_admin_update" on public.apps_page_settings;
create policy "apps_page_admin_update" on public.apps_page_settings
  for update to authenticated
  using (public.is_aboud_admin()) with check (public.is_aboud_admin());

grant select on public.apps_page_settings to anon, authenticated;
grant insert, update on public.apps_page_settings to authenticated;

-- ---------------------------------------------------------------------
-- 4) تخزين الصور (Supabase Storage) — قراءة عامة، رفع/تعديل/حذف للمسؤول فقط
--    الأنواع المسموحة: PNG / JPG / WebP / GIF / AVIF (بدون SVG عمداً)، حتى 5MB للملف.
-- ---------------------------------------------------------------------
insert into storage.buckets (id, name, public, file_size_limit, allowed_mime_types)
values (
  'app-media', 'app-media', true, 5242880,
  array['image/png', 'image/jpeg', 'image/webp', 'image/gif', 'image/avif']
)
on conflict (id) do update
  set public = excluded.public,
      file_size_limit = excluded.file_size_limit,
      allowed_mime_types = excluded.allowed_mime_types;

drop policy if exists "app_media_admin_select" on storage.objects;
create policy "app_media_admin_select" on storage.objects
  for select to authenticated
  using (bucket_id = 'app-media' and public.is_aboud_admin());

drop policy if exists "app_media_admin_insert" on storage.objects;
create policy "app_media_admin_insert" on storage.objects
  for insert to authenticated
  with check (bucket_id = 'app-media' and public.is_aboud_admin());

drop policy if exists "app_media_admin_update" on storage.objects;
create policy "app_media_admin_update" on storage.objects
  for update to authenticated
  using (bucket_id = 'app-media' and public.is_aboud_admin())
  with check (bucket_id = 'app-media' and public.is_aboud_admin());

drop policy if exists "app_media_admin_delete" on storage.objects;
create policy "app_media_admin_delete" on storage.objects
  for delete to authenticated
  using (bucket_id = 'app-media' and public.is_aboud_admin());
